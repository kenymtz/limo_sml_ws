cmake_minimum_required(VERSION 3.8)
project(limo_nav_offline_lab)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
set(deps rclcpp dwb_core dwb_msgs nav_2d_msgs nav_2d_utils nav2_navfn_planner nav2_core nav2_costmap_2d nav2_util nav_msgs geometry_msgs tf2_ros)
foreach(d ${deps})
  find_package(${d} REQUIRED)
endforeach()
add_executable(dwb_offline_eval src/dwb_offline_eval.cpp)
target_compile_features(dwb_offline_eval PUBLIC cxx_std_17)
ament_target_dependencies(dwb_offline_eval ${deps})
add_executable(dwb_closed_loop_sim src/dwb_closed_loop_sim.cpp)
target_compile_features(dwb_closed_loop_sim PUBLIC cxx_std_17)
ament_target_dependencies(dwb_closed_loop_sim ${deps})
install(TARGETS dwb_offline_eval dwb_closed_loop_sim DESTINATION lib/${PROJECT_NAME})
ament_package()
